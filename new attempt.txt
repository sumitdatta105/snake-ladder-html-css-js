<script>
const snakes = {99:78,89:68,84:64,62:41,52:29,38:18,33:7};
const ladders = {3:22,8:30,20:41,28:55,36:65,51:71,73:93};

const wrap = document.getElementById("wrap");
const grid = document.getElementById("grid");
const turnText = document.getElementById("turnText");
const diceText = document.getElementById("dice");

const diceSound = document.getElementById("diceSound");
const snakeSound = document.getElementById("snakeSound");
const ladderSound = document.getElementById("ladderSound");
const winSound = document.getElementById("winSound");

let size=600, cell=size/10, positions={};
let p1=0,p2=0,entered1=false,entered2=false,turn=1;
let name1="",name2="";

/* GRID */
function buildGrid(){
  for(let r=0;r<10;r++){
    for(let c=0;c<10;c++){
      let index=r*10+(r%2===0?c+1:10-c);
      let d=document.createElement("div");
      grid.appendChild(d);
      positions[index]={x:c*cell+cell/2,y:r*cell+cell/2};
    }
  }
}

/* PLAYERS */
function placePlayers(){
  document.querySelectorAll(".player").forEach(p=>p.remove());

  [[p1,"p1",1],[p2,"p2",2]].forEach((pl,i)=>{
    let d=document.createElement("div");
    d.className="player "+pl[1];
    d.textContent=pl[2];

    let pos=pl[0]?positions[pl[0]]:{x:i?size-30:30,y:size+30};
    d.style.left=pos.x+"px";
    d.style.top=pos.y+"px";
    wrap.appendChild(d);
  });
}

function updateTurn(){
  turnText.textContent=(turn===1?name1:name2)+"'s Turn";
}

/* START */
startBtn.onclick=()=>{
  name1=name1Input.value||"Player 1";
  name2=name2Input.value||"Player 2";
  startScreen.style.display="none";
  gameContainer.style.display="flex";
  buildGrid();
  placePlayers();
  updateTurn();
};

/* DICE */
rollBtn.onclick=()=>{
  let roll=Math.floor(Math.random()*6)+1;

  diceSound.currentTime=0;
  diceSound.play();

  diceText.textContent="Dice: "+roll;

  function move(pos,entered){
    if(!entered){
      if(roll===1) return {p:1,e:true};
      return {p:pos,e:false};
    }

    if(pos+roll<=100) pos+=roll;

    if(snakes[pos]){
      pos=snakes[pos];
      snakeSound.play();
    }
    else if(ladders[pos]){
      pos=ladders[pos];
      ladderSound.play();
    }

    return {p:pos,e:true};
  }

  if(turn===1){
    let r=move(p1,entered1);
    p1=r.p; entered1=r.e;
    if(p1===100){
      winSound.play();
      alert(name1+" wins!");
      location.reload();
    }
    turn=2;
  } else {
    let r=move(p2,entered2);
    p2=r.p; entered2=r.e;
    if(p2===100){
      winSound.play();
      alert(name2+" wins!");
      location.reload();
    }
    turn=1;
  }

  placePlayers();
  updateTurn();
};
